**Goal:** Close two confirmed availability/DoS gaps in the Hub session-verification gate (`src/interface/lib/hubAccess.js`) without weakening its fail-closed security posture.

**Architecture:** Both fixes are localized to `hubAccess.js`'s internal caching logic — no new files, no new dependencies, no change to the public `criarMiddlewareHubAccess()` signature or to `server.js`. **Tech Stack:** Node.js built-ins only (`node:crypto`, `node:test`, `node:assert/strict`) — matches the existing file, no new dependency.

**Spec:** N/A — this plan implements two findings from an ad-hoc adversarial security review (skill `adversarial-security-testing`), not a written design spec. See the "Findings" section below in lieu of a spec.

```javascript
const JWKS_COOLDOWN_MS = 5000;
```

```javascript
  async function buscarJwks() {
    if (fetchJwksEmVoo) return fetchJwksEmVoo; // single-flight: reaproveita o fetch já em andamento
    fetchJwksEmVoo = (async () => {
      const resposta = await fetchImpl(jwksUrl, { signal: AbortSignal.timeout(TIMEOUT_MS) });
      if (!resposta.ok) throw new Error(`JWKS do hub indisponível: HTTP ${resposta.status}`);
      const corpo = await resposta.json();
      chaves = new Map((corpo.keys ?? []).map((jwk) => [jwk.kid, jwk]));
      ultimoFetchJwksEm = Date.now();
    })();
    try {
      await fetchJwksEmVoo;
    } finally {
      fetchJwksEmVoo = null;
    }
  }

  async function obterChave(kid) {
    if (!chaves.has(kid)) {
      // `kid` desconhecido pode vir de um atacante testando valores aleatórios — sem esse
      // cooldown, cada requisição não autenticada gera um fetch real ao hub (amplificação).
      // Um refresh legítimo de chave (rotação no hub) só demora no máximo JWKS_COOLDOWN_MS
      // a mais pra ser enxergado, o que é aceitável.
      const dentroDoCooldown = Date.now() - ultimoFetchJwksEm < JWKS_COOLDOWN_MS;
      if (!dentroDoCooldown) await buscarJwks();
    }
    return chaves.get(kid) ?? null;
  }
```

```javascript
  let hubDisponivel = true;
```

```javascript
  async function versaoAindaValida(sub, ver) {
    const chave = `${sub}:${ver}`;
    const agora = Date.now();
    const cache = cacheVersao.get(chave);
    if (cache && cache.expiraEm > agora) return cache.valido;

    try {
      const url = `${hubUrl}/api/verificar?usuario=${encodeURIComponent(sub)}&versao=${ver}`;
      const resposta = await fetchImpl(url, { signal: AbortSignal.timeout(TIMEOUT_MS) });
      const corpo = await resposta.json();
      const valido = resposta.ok && corpo?.valido === true;
      // `confirmadoEm` marca a última vez que tivemos uma resposta REAL do hub (não reaproveitada)
      // — é o que limita por quanto tempo total uma indisponibilidade pode mascarar uma revogação.
      cacheVersao.set(chave, { valido, expiraEm: agora + CACHE_VERSAO_MS, confirmadoEm: agora });
      return valido;
    } catch {
      if (cache && agora - cache.confirmadoEm < TETO_REAPROVEITAMENTO_MS) {
        cacheVersao.set(chave, {
          valido: cache.valido,
          expiraEm: agora + CACHE_VERSAO_MS,
          confirmadoEm: cache.confirmadoEm, // nunca avança sem uma confirmação real
        });
        return cache.valido;
      }
      // Sem cache prévio, ou teto de reaproveitamento estourado durante uma indisponibilidade
      // contínua: falha fechado em vez de confiar num valor conhecido havia tempo demais.
      return false;
    }
  }
```
